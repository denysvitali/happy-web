// so the vectors come from the real Go implementation. Never committed there.
package cryptoutil

import (
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"
)

func TestHappyWebVectors(t *testing.T) {
	out := os.Getenv("HAPPY_WEB_VECTORS_OUT")
	if out == "" {
		t.Skip("HAPPY_WEB_VECTORS_OUT not set")
	}
	secret := make([]byte, 32)
	for i := range secret {
		secret[i] = byte(i + 1)
	}
	seed := flutterDeriveContentKey(secret)
	pub, _ := flutterSeedKeyPair(seed)

	dataKey := make([]byte, 32)
	for i := range dataKey {
		dataKey[i] = byte(0xA0 + i)
	}
	payload := map[string]any{"role": "user", "content": map[string]any{"type": "text", "text": "héllo ✓"}}

	var legacyKey [32]byte
	copy(legacyKey[:], secret)
	legacy, err := EncryptLegacyJSON(payload, &legacyKey)
	if err != nil {
		t.Fatal(err)
	}
	gcm, err := EncryptDataKeyJSON(payload, dataKey)
	if err != nil {
		t.Fatal(err)
	}
	sealed, err := EncryptForPublicKey(dataKey, pub[:])
	if err != nil {
		t.Fatal(err)
	}
	wrapped := append([]byte{0}, sealed...)
	urlSample := []byte{0xfb, 0xff, 0xfe, 0x00, 0x01}

	v := map[string]any{
		"masterSecretHex":     hex.EncodeToString(secret),
		"contentSeedHex":      hex.EncodeToString(seed),
		"contentPublicKeyHex": hex.EncodeToString(pub[:]),
		"payload":             payload,
		"legacyCiphertextB64": EncodeBase64(legacy),
		"dataKeyHex":          hex.EncodeToString(dataKey),
		"dataKeyCiphertextB64": EncodeBase64(gcm),
		"wrappedDataKeyB64":   EncodeBase64(wrapped),
		"base64UrlSampleHex":  hex.EncodeToString(urlSample),
		"base64UrlSample":     EncodeBase64URL(urlSample),
	}
	b, err := json.MarshalIndent(v, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
